\recipe{%
	fullpage=false,
	title = {Scrambled Eggs (List)},
	serves = {2},
	preptime = {5 min},
	origin = {Classic},
	ingredientstyle=list,
	ingredients={%
			\ingredientsection{Ingredients}
			\ingredient{eggs}[4][][large]
			\ingredient{butter}[1][tbsp]
			\ingredient{milk}[2][tbsp]
			\ingredient{salt and pepper}[][][to taste]

			\ingredientsection{Serving}
			\ingredient{sliced bread}[2][slices]
			\ingredient{butter}[1][tbsp]
		},
	instructions={%
			\instruction Crack eggs into a bowl, add milk, salt, and pepper. Whisk until combined.
			\instruction Melt butter in a non-stick pan over medium heat.
			\instruction Pour in egg mixture and gently stir with a spatula until just set.
		},
}
